Unoptimized script:
WITH customer\_rental\_counts AS (
    SELECT
        c.customer\_id,
        c.first\_name,
        c.last\_name,
        COUNT(r.rental\_id) AS rental\_count
    FROM customer AS c
    JOIN rental AS r
        ON c.customer\_id = r.customer\_id
    GROUP BY
        c.customer\_id,
        c.first\_name,
        c.last\_name
)
SELECT
    customer\_id,
    first\_name,
    last\_name,
    rental\_count
FROM customer\_rental\_counts
WHERE rental\_count > (
    SELECT AVG(rental\_count)
    FROM customer\_rental\_counts
)
ORDER BY rental\_count DESC;

Optimized script:
WITH rental\_counts AS (
    SELECT
        customer\_id,
        COUNT(\*) AS rental\_count
    FROM rental
    GROUP BY customer\_id
),
average\_count AS (
    SELECT AVG(rental\_count) AS avg\_rental\_count
    FROM rental\_counts
)
SELECT
    c.customer\_id,
    c.first\_name,
    c.last\_name,
    rc.rental\_count
FROM rental\_counts AS rc
JOIN customer AS c
    ON c.customer\_id = rc.customer\_id
CROSS JOIN average\_count AS ac
WHERE rc.rental\_count > ac.avg\_rental\_count
ORDER BY rc.rental\_count DESC;
